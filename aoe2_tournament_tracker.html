⚙️ Python Fetch Script — Pull Live Data from Liquipedia
Run this from your own machine (not Claude's sandbox — Liquipedia blocks server-side requests without a real User-Agent). The script respects Liquipedia's rate limits (1 req/2s). After it runs, paste the JSON output into the panel above.

Setup: pip install requests then run the script. Replace the User-Agent email with your own contact info as Liquipedia requires. 
import requests
import json
import time

# ── Liquipedia requires a descriptive User-Agent with contact info ──
HEADERS = {
    "User-Agent": "AoE2TournamentTracker/1.0 (your-email@example.com) personal-research-tool",
    "Accept-Encoding": "gzip",
}

BASE_URL = "https://liquipedia.net/ageofempires/api.php"

PRINTOUTS = [
    "Liquipedia tier",
    "Start date",
    "End date",
    "Prize pool",
    "Prize pool currency",
    "Region",
    "Type",
    "First place",
    "Second place",
    "Third place",
    "Number of participants",
    "Organizer",
]

def fetch_tournaments(limit=50, offset=0, start_after="2025-01-01"):
    params = {
        "action": "askargs",
        "format": "json",
        "conditions": f"[[Category:Tournaments]][[Game::Age of Empires II]][[Start date::>{start_after}]]",
        "printouts": "|".join(PRINTOUTS),
        "parameters": f"sort=Start date|order=desc|limit={limit}|offset={offset}",
    }
    # action=askargs: max 1 request per 2 seconds — enforced, bans issued for violations
    time.sleep(2)
    response = requests.get(BASE_URL, params=params, headers=HEADERS, timeout=15)
    response.raise_for_status()
    return response.json()

def fetch_all(max_results=100, start_after="2025-01-01"):
    all_results = {}
    offset = 0
    limit = 50
    while offset < max_results:
        print(f"Fetching offset={offset}…")
        batch = fetch_tournaments(limit=min(limit, max_results - offset), offset=offset, start_after=start_after)
        results = batch.get("query", {}).get("results", {})
        if not results:
            break
        all_results.update(results)
        if len(results) < limit:
            break
        offset += limit

    return {"query": {"results": all_results}}

# ── Run ──
print("Fetching AoE2 tournaments from Liquipedia…")
data = fetch_all(max_results=100, start_after="2025-01-01")
count = len(data["query"]["results"])
print(f"Fetched {count} tournaments.")

# ── Save and print ──
with open("liquipedia_aoe2_tournaments.json", "w", encoding="utf-8") as f:
    json.dump(data, f, indent=2, ensure_ascii=False)
